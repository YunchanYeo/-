# 2) DNS A 레코드가 이 ECS 공인 IP 를 가리키는지 확인 후 저장
# 3) cd deploy/china-test && docker compose restart caddy
# 4) backend/.env 의 API_PUBLIC_BASE_URL=https://api.example.com
# 5) frontend/config/index.js 의 CLOUD_HTTPS_API_BASE_OVERRIDE
#

		protocols tls1.2 tls1.3
	}
